Bài 12. Xây dựng một Automation Framework nhỏ

Mục tiêu: Hợp nhất mọi thứ đã học thành một dự án nhỏ, có cấu trúc rõ ràng, chạy được bằng mvnw, sẵn sàng để đưa lên Git và show trong phỏng vấn.

Kiến thức cần dùng: Maven + Wrapper, TestNG, POM, Explicit Wait, Config, DriverFactory, @DataProvider, screenshot khi fail, báo cáo Surefire.

Code gợi ý: Config + BaseTest rút gọn

java
// config/Config.java
public final class Config {
    private static final Properties P = new Properties();
    static {
        try (InputStream in = Config.class.getClassLoader().getResourceAsStream("config.properties")) {
            P.load(in);
        } catch (IOException | NullPointerException e) {
            throw new IllegalStateException("Thiếu config.properties", e);
        }
    }
    private Config() {}
    public static String get(String k, String d) { return System.getProperty(k, P.getProperty(k, d)); }
    public static String browser()    { return get("browser", "chrome"); }
    public static boolean headless()  { return Boolean.parseBoolean(get("headless", "false")); }
    public static String baseUrl()    { return get("base.url", "https://the-internet.herokuapp.com"); }
    public static Duration timeout()  { return Duration.ofSeconds(Long.parseLong(get("timeout.seconds", "15"))); }
}
java
// base/BaseTest.java
public abstract class BaseTest {
    private static final ThreadLocal<WebDriver> DRIVER = new ThreadLocal<>();
    protected WebDriver driver() { return DRIVER.get(); }

    @BeforeMethod
    public void setup() {
        DRIVER.set(DriverFactory.create(Config.browser(), Config.headless()));
        driver().manage().window().maximize();
    }

    @AfterMethod(alwaysRun = true)
    public void tearDown(ITestResult result) {
        if (driver() != null) {
            if (result.getStatus() == ITestResult.FAILURE) Screenshots.take(driver(), result.getName());
            driver().quit();
            DRIVER.remove();
        }
    }
}
