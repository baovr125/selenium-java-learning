Phần B. Viết test

java
package tests;

import base.BaseTest;
import org.openqa.selenium.By;
import org.openqa.selenium.WebElement;
import org.testng.Assert;
import org.testng.annotations.Test;

import java.util.List;

public class LocatorPracticeTest extends BaseTest {

    @Test
    public void allLocatorsPointToSameElement() {
        driver.get("https://the-internet.herokuapp.com/login");

        WebElement byId   = driver.findElement(By.id("username"));
        WebElement byName = driver.findElement(By.name("username"));
        WebElement byCss  = driver.findElement(By.cssSelector("input#username"));
        WebElement byXpath = driver.findElement(By.xpath("//input[@name='username']"));

        Assert.assertEquals(byName, byId, "name phải trỏ cùng phần tử với id");
        Assert.assertEquals(byCss, byId);
        Assert.assertEquals(byXpath, byId);
    }

    @Test
    public void linkLocators() {
        driver.get("https://the-internet.herokuapp.com/");
        WebElement full    = driver.findElement(By.linkText("Checkboxes"));
        WebElement partial = driver.findElement(By.partialLinkText("Checkbox"));
        Assert.assertEquals(partial, full);
    }

    @Test
    public void countAndTagName() {
        driver.get("https://the-internet.herokuapp.com/tables");
        List<WebElement> rows = driver.findElements(By.cssSelector("#table1 tbody tr"));
        Assert.assertEquals(rows.size(), 4, "Example 1 có 4 dòng dữ liệu");

        List<WebElement> cellsOfFirstRow = rows.get(0).findElements(By.tagName("td"));   // tìm CON của hàng
        Assert.assertTrue(cellsOfFirstRow.size() >= 5);
    }

    @Test
    public void xpathAxesAndPosition() {
        driver.get("https://the-internet.herokuapp.com/tables");
        String lastRowLastName = driver.findElement(By.xpath("//table[@id='table1']/tbody/tr[last()]/td[1]")).getText();
        String nextToLast     = driver.findElement(By.xpath("//table[@id='table1']/tbody/tr[last()-1]/td[1]")).getText();
        System.out.println("Hàng cuối: " + lastRowLastName + " | Kế cuối: " + nextToLast);

        // đi từ ô email lên hàng cha rồi lấy ô Last Name của cùng hàng
        String owner = driver.findElement(
                By.xpath("//td[text()='jsmith@gmail.com']/preceding-sibling::td[2]")).getText();
        System.out.println("Chủ email jsmith@gmail.com: " + owner);
    }

    @Test
    public void missingElementBehaviour() {
        driver.get("https://the-internet.herokuapp.com/");
        Assert.assertTrue(driver.findElements(By.id("khong-ton-tai")).isEmpty(), "findElements trả list rỗng");
        // driver.findElement(By.id("khong-ton-tai"));  // bỏ comment để xem NoSuchElementException
    }
}

Phần C. Thử thách

  1. Viết CSS và XPath cho: link "Elemental Selenium" ở chân trang; checkbox đầu tiên ở /checkboxes; nút "Add Element" ở /add_remove_elements/.
  2. Viết XPath lấy email của người có Last Name = "Doe" trong bảng (dùng anh em following-sibling).
  3. Cố tình viết By.className("flash success") và đọc exception; sửa lại bằng CSS .flash.success (sau khi login thành công).
  4. Mở /hovers: lấy toàn bộ h5 bằng findElements, in text (lưu ý ẩn ⇒ rỗng) rồi chọn cách đúng (xem Ex5).

Kết quả mong đợi

  • 4 test của Phần B pass.
  • Bạn giải thích được vì sao nên ưu tiên id/CSS, khi nào cần XPath.
